Tarea 2. Protective Put [MATRICULA REDACTADA]

Detalles de las Posiciones

shares <- 100
entry_price <- 335.72
strike_price <- 335.00
put_price <- 4.05

# Rango de precios de AAPL al vencimiento

future_price_range <- seq(from = 0, to = 600, by = 1)

Análisis de ganancia/pérdida

# Ganancia o pérdida de las 100 acciones

long_stock_profit <- 
  (future_price_range - entry_price) * shares


# Ganancia o pérdida de la Put comprada

put_buyer_profit <- 
  (pmax(strike_price - future_price_range, 0) - put_price) * shares


# Ganancia o pérdida total del Protective Put

protective_put <- 
  long_stock_profit + put_buyer_profit

Indicadores de la estrategia

# Punto de equilibrio

breakeven <- entry_price + put_price


# Pérdida máxima

max_loss <- 
  (strike_price - entry_price - put_price) * shares

Gráfica Protective Put

# Configuración de la gráfica

par(mar = c(5.1, 4.1, 4.1, 8.1))

y_min <- min(c(
  protective_put,
  long_stock_profit,
  put_buyer_profit
))

y_max <- max(c(
  protective_put,
  long_stock_profit,
  put_buyer_profit
))

y_pad <- 0.08 * (y_max - y_min)


# Gráfica del Protective Put

plot(
  x = future_price_range,
  y = protective_put,
  type = "l",
  lwd = 2,
  main = paste0(
    "Protective Put AAPL (S0 = ",
    entry_price,
    ", K = ",
    strike_price,
    ", Put = ",
    put_price,
    ")"
  ),
  xlab = "Precio de AAPL al vencimiento (ST)",
  ylab = "Ganancia / Pérdida total (USD)",
  ylim = c(
    y_min - y_pad,
    y_max + y_pad
  )
)

# Línea de las acciones

lines(
  x = future_price_range,
  y = long_stock_profit,
  col = "blue",
  lty = 2,
  lwd = 2
)


# Línea de la Put

lines(
